#library "GRUL"
#include "zcommon.acs"

// Состояние относится только к текущей карте.
int team1_frags;
int team2_frags;

int deathmatch_enabled;
int deathmatch_limit;
int deathmatch_finished;
int deathmatch_winner;


// ------------------------------------------------------------
// ОСНОВНОЙ ЦИКЛ
// ------------------------------------------------------------

script 999 OPEN
{
    while (TRUE)
    {
        if (deathmatch_enabled && !deathmatch_finished)
        {
            CheckDeathmatchVictory();
        }

        Delay(35);
    }
}


// ------------------------------------------------------------
// ДИРЕКТОР
// ------------------------------------------------------------

script 700 (int requestedLimit)
{
    // Защита от нескольких директоров на карте.
    if (deathmatch_enabled)
    {
        terminate;
    }

    deathmatch_enabled = TRUE;
    deathmatch_finished = FALSE;
    deathmatch_winner = 0;

    team1_frags = 0;
    team2_frags = 0;

    if (requestedLimit > 0)
    {
        deathmatch_limit = requestedLimit;
    }
    else
    {
        deathmatch_limit = 50;
    }
}


// ------------------------------------------------------------
// HUD ДЛЯ КАЖДОГО ИГРОКА
// ------------------------------------------------------------

script 701 ENTER
{
    while (TRUE)
    {
        if (deathmatch_enabled)
        {
            SetFont("BIGFONT");

            HudMessage(
                s:"TEAM 1: ",
                i:team1_frags,
                s:" / ",
                i:deathmatch_limit;

                HUDMSG_PLAIN,
                100,
                CR_CYAN,
                0.38,
                0.05,
                2.0
            );

            HudMessage(
                s:"TEAM 2: ",
                i:team2_frags,
                s:" / ",
                i:deathmatch_limit;

                HUDMSG_PLAIN,
                101,
                CR_RED,
                0.62,
                0.05,
                2.0
            );

            if (deathmatch_finished)
            {
                if (deathmatch_winner == 1)
                {
                    HudMessage(
                        s:"TEAM 1 WINS!";

                        HUDMSG_PLAIN,
                        102,
                        CR_CYAN,
                        0.5,
                        0.30,
                        2.0
                    );
                }
                else if (deathmatch_winner == 2)
                {
                    HudMessage(
                        s:"TEAM 2 WINS!";

                        HUDMSG_PLAIN,
                        102,
                        CR_RED,
                        0.5,
                        0.30,
                        2.0
                    );
                }
            }
        }

        Delay(35);
    }
}


// ------------------------------------------------------------
// СЧЁТЧИКИ СМЕРТЕЙ
// ------------------------------------------------------------

script 1 (void)
{
    if (!deathmatch_enabled || deathmatch_finished)
    {
        terminate;
    }

    team1_frags++;
    CheckDeathmatchVictory();
}


script 2 (void)
{
    if (!deathmatch_enabled || deathmatch_finished)
    {
        terminate;
    }

    team2_frags++;
    CheckDeathmatchVictory();
}


// ------------------------------------------------------------
// ПРОВЕРКА ПОБЕДЫ
// ------------------------------------------------------------

function void CheckDeathmatchVictory(void)
{
    if (!deathmatch_enabled || deathmatch_finished)
    {
        return;
    }

    if (team1_frags >= deathmatch_limit)
    {
        deathmatch_finished = TRUE;
        deathmatch_winner = 1;

        ACS_ExecuteAlways(501, 0, 0, 0, 0);
    }
    else if (team2_frags >= deathmatch_limit)
    {
        deathmatch_finished = TRUE;
        deathmatch_winner = 2;

        ACS_ExecuteAlways(502, 0, 0, 0, 0);
    }
}


// ------------------------------------------------------------
// ЗАВЕРШЕНИЕ DEATHMATCH
// ------------------------------------------------------------

script 501 (void)
{
    Delay(35 * 6);
    Exit_Normal(0);
}


script 502 (void)
{
    Delay(35 * 6);
    Exit_Normal(0);
}
