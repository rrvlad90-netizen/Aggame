ACTOR ASpan : Inventory {
    Inventory.MaxAmount 1
	+INVENTORY.UNDROPPABLE
}

ACTOR BodyOfBot
{
  Health 100
  Radius 20
  Height 20
  Speed 0
  Monster
  +NOPAIN
  +FRIENDLY
	+LOOKALLAROUND
  SeeSound "grunt/sight"
  AttackSound "grunt/attack"
  PainSound "grunt/pain"
  DeathSound "grunt/death"
  ActiveSound "grunt/active"
  Obituary "$OB_ZOMBIE"
  DropItem "Clip"
	Mass 9999
  States
  {
  Spawn:
    //POSS A 0 A_Warp(AAPTR_TARGET, 0, 0, 56, WARPF_NOCHECKPOSITION)
	POSS A 0
    POSS A 0 Thing_Hate(0, 99, 0)  //    POSS A 0 Thing_Hate(0, 99, 6)
  Spn:
    TNT1 A 0 A_Warp(AAPTR_MASTER, 0, 0, 56, 0)
    POSS A 1 A_Look
    Loop

  See:
    POSS A 1 
	{
	A_FaceTarget();
	A_Chase("Missile","Missile", CHF_DONTMOVE);
	A_Warp(AAPTR_MASTER, 0, 0, 56, WARPF_NOCHECKPOSITION);
	}
    Loop

  Missile:
    POSS E 1
	{	
	A_FaceTarget;
	A_Warp(AAPTR_MASTER, 0, 0, 56, WARPF_NOCHECKPOSITION);
	}
    POSS F 1
    {	
	A_FaceTarget;
	A_PosAttack;
	A_Warp(AAPTR_MASTER, 0, 0, 56, WARPF_NOCHECKPOSITION);
	}
    POSS E 1
	{
	A_FaceTarget;
	A_Warp(AAPTR_MASTER, 0, 0, 56, WARPF_NOCHECKPOSITION);
	}
    Goto See

  Death:
    TNT1 A 0 A_KillMaster
    POSS H 5
    POSS I 5 A_Scream
    POSS J 5 A_NoBlocking
    POSS K 5
    POSS L 1 A_FadeOut
    Stop
  }
}


ACTOR LegsofBot replaces Demon
{
  Health 100
  Radius 20
  Height 56
  Speed 8
  PainChance 255
  Monster
  +SLIDESONWALLS
  +FLOORCLIP
  +FRIENDLY
  +NOPAIN
  SeeSound "grunt/sight"
  AttackSound "grunt/attack"
  PainSound "grunt/pain"
  DeathSound "grunt/death"
  ActiveSound "grunt/active"
  Obituary "$OB_ZOMBIE"
	Mass 9999
  States
  {
  Spawn:
    //POSS A 1 NoDelay A_SpawnItem("BodyOfBot", 0, 56)
	POSS A 0 Thing_ChangeTID(0, 88)
    POSS A 0 Thing_Hate(0, 199, 6)
    POSS A 1 A_JumpIfInventory("ASpan",1,"Spn")
    Goto Diss
  Spn:
    POSS A 1 A_SpawnItemEx("BodyOfBot", 0, 0, 56, 0, 0, 0, 0, SXF_SETMASTER | SXF_TRANSFERPOINTERS, 0)
  	//POSS A 1 A_SpawnItem("BodyOfBot", 0, 56)
	POSS A 20
    POSS A 3 A_Look
    Loop

  See:
    POSS AABBCCDD 1 A_Chase
    Loop

  Pain:
    POSS G 1
    Goto See

  Death:
    TNT1 A 0 A_KillChildren
    POSS H 5
    POSS I 5 A_Scream
    POSS J 5 A_NoBlocking
    POSS K 5
    POSS L 250
    POSS L 5 A_Respawn
    Stop

 Diss: 
    TNT1 A 0 GiveInventory("ASpan",1)
    TNT1 A 5 A_ClearTarget
    TNT1 A 5 A_Respawn
    Stop
  }
}
