ACTOR BodyBase
{
  Health 100
  Radius 1
  Height 1
  Speed 0
  Monster
  +NOPAIN
	//-FASTER
	//-DONTHURTSPECIES
	-PUSHABLE
	-SOLID
	+LOOKALLAROUND
  SeeSound "grunt/sight"
  AttackSound "grunt/attack"
  PainSound "grunt/pain"
  DeathSound "grunt/death"
  ActiveSound "grunt/active"
  Obituary "$OB_ZOMBIE"
  DropItem "Clip"
  Mass 9999

  States
  {
  Spawn:
    //POSS A 0 A_Warp(AAPTR_TARGET, 0, 0, 56, WARPF_NOCHECKPOSITION)
	POSS A 0
    //POSS A 0 Thing_Hate(0, 88, 3)
  Spn:
    TNT1 A 0 A_Warp(AAPTR_MASTER, 0, 0, 56, WARPF_NOCHECKPOSITION)
    POSS A 1 A_Look
    Loop

  See:
    POSS A 1 
	{
	A_FaceTarget(90);
	A_Chase("Missile","Missile", CHF_DONTMOVE);
	A_Warp(AAPTR_MASTER, 0, 0, 56, 0, WARPF_USECALLERANGLE | WARPF_COPYPITCH);
	}
    Loop

  Missile:
    POSS E 1
	{	
	A_FaceTarget;
	A_Warp(AAPTR_MASTER, 0, 0, 56, 0, WARPF_USECALLERANGLE | WARPF_COPYPITCH);
	}
    POSS F 1
    {	
	A_FaceTarget;
	A_PosAttack;
	A_Warp(AAPTR_MASTER, 0, 0, 56, 0, WARPF_USECALLERANGLE | WARPF_COPYPITCH);
	}
    POSS EEE 1
	{
	A_FaceTarget;
	A_Warp(AAPTR_MASTER, 0, 0, 56, 0, WARPF_USECALLERANGLE | WARPF_COPYPITCH);
	}
    Goto See

  Death:
    TNT1 A 0 A_ClearTarget
    TNT1 A 0 A_KillMaster
    POSS H 5
    POSS I 5 A_Scream
    POSS J 5 A_NoBlocking
    POSS K 5
    POSS L 1 A_FadeOut
    Stop
  }
}

  +SLIDESONWALLS
  +FLOORCLIP
  SeeSound "grunt/sight"
  AttackSound "grunt/attack"
  PainSound "grunt/pain"
  DeathSound "grunt/death"
  ActiveSound "grunt/active"
  Obituary "$OB_ZOMBIE"
  Mass 9999

  States
  {
  Spawn:
    //POSS A 1 NoDelay A_SpawnItem("BodyOfEnemy", 0, 56)
	//POSS A 0 Thing_ChangeTID(0, 99)
    //POSS A 0 Thing_Hate(0, 199, 6)
    POSS A 1 A_JumpIfInventory("ASpan",1,"Spn")
    Goto Diss
  Spn:
    POSS A 1 A_SpawnItemEx("BodyOfEnemy", 0, 0, 56, 0, 0, 0, 0, SXF_SETMASTER | SXF_TRANSFERPOINTERS, 0)
	POSS A 20
    POSS A 3 A_Look
    Loop

  See:
    POSS AABBCCDD 1 A_Chase
    Loop

  Pain:
  TNT1 A 0
  Goto See
  
  StrifeLeft:
  POSS GGGG 1
	{  
	A_FaceTarget;
	ThrustThing(angle*256/360+210,5, 0, 0);
	}
  StrifeRight:
    POSS GGGG 1
	{  
	A_FaceTarget;
	ThrustThing(angle*256/360+210,-5, 0, 0);
	}
    Goto See

  Death:
    TNT1 A 0 A_KillChildren
	TNT1 A 0 A_ClearTarget
    POSS H 5
    POSS I 5 A_Scream
    POSS J 5 A_NoBlocking
    POSS K 5
    POSS L 250
    POSS L 5 A_Respawn
    Stop

 Diss: 
    TNT1 A 0 GiveInventory("ASpan",1)
    TNT1 A 5 A_ClearTarget
    TNT1 A 5 A_Respawn
    Stop
  }
}
