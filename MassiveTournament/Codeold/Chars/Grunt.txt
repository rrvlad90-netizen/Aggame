ACTOR Grunt : HMBasePawn
{
	Player.DisplayName "Grunt"
	Player.SoundClass "Q2Guy"
	Player.Portrait "PROFGRNT"
	States
	{
	Spawn:
	  PLAY Z 1
      Loop
    See:
      PLAY AB 2
	  PLAY C 2 A_SpawnItemEx("FootStep",0,0,0,0,0,-8)
	  PLAY DEFG 2
	  PLAY H 2 A_SpawnItemEx("FootStep",0,0,0,0,0,-8)
	  PLAY I 2
      Loop
	See.Walk:
      PLAW ABCDEFGHIJKL 2
      Loop
	RunBack.Walk:
	  PLAW LKJIHGFEDCBA 2
      Loop
	RunBack:
      PLAY KL 2
	  PLAY M 2 A_SpawnItemEx("FootStep",0,0,0,0,0,-8)
	  PLAY NOPQ 2
	  PLAY R 2 A_SpawnItemEx("FootStep",0,0,0,0,0,-8)
	  PLAY ST 2
      Loop
    JumpLoop:
      PLYJ ABCDEFGH 2 A_CheckFloor("Land")
	  PLYJ I 0 A_JumpIf(waterlevel>=2,"Swim")
	  PLYJ I 1 A_CheckFloor("Land")
      Goto JumpLoop+8
	JumpLoop.Back:
      PLJ2 ABCDEFGHI 2 A_CheckFloor("Land")
	  PLJ2 J 0 A_JumpIf(waterlevel>=2,"Swim")
	  PLJ2 J 1 A_CheckFloor("Land")
      Goto JumpLoop+9
    Land: 
	  PLYJ M 2 A_SetUserVar("user_landed",1)
	  PLYJ NOP 2 A_JumpIf(momz<-6,"JumpLoop")
	  TNT1 A 0 A_SetUserVar("user_isrunningforward",0)
	  TNT1 A 0 A_SetUserVar("user_iswalkingforward",0)
	  TNT1 A 0 A_SetUserVar("user_isrunningback",0)
	  TNT1 A 0 A_SetUserVar("user_iswalkingback",0)
	  TNT1 A 0 A_SetUserVar("user_isplayerbusy",0)
      Goto Spawn
	Swim:
      PLYS ABCDEFGHIJ 3 A_JumpIf(waterlevel<=1,"JumpLoop")
      Loop
	AnimTaunt:
	  PLAY Z 0 A_PlaySound("Q2TAUNT",2)
	  PLAY Z 1 A_SetUserVar("user_istaunting",0)
	  Goto Spawn
	Death:
	  PLAY Z 0 A_PlaySound("world/quake",5)
	  PLAY Z 0 A_PlaySound("world/quake",6)
	  PLAY Z 0 A_PlaySound("world/quake",7)
	  PLAY Z 0 A_Jump(170,"Death1","Death2")
	  NADD A 2 A_PlayerScream
	  NADD B 2 A_NoBlocking
	  NADD CDEFGHIJKLMNOPQRSTUVWXYZ 2
	  NADE ABCD 2
	  NADE E -1
	  Stop
	Death1:
	  PLD2 A 2 A_PlayerScream
	  PLD2 B 2 A_NoBlocking
	  PLD2 CDEFGHIJKLMNOPQRSTUVWXYZ 2
	  PLE2 ABC 2
	  PLE2 D -1
	  Stop
	Death2:
	  PLD3 A 2 A_PlayerScream
	  PLD3 B 2 A_NoBlocking
	  PLD3 CDEFGHIJKLMNOPQRSTUVWXYZ 2
	  PLE3 ABC 2
	  PLE3 D -1
	  Stop
	}
}
