ACTOR Sorlag : HMBasePawnHeavy
{
	Health 175
	Radius 16
	Height 104
	RenderRadius 40
	Mass 200
	Player.DisplayName "Sorlag"
	Player.SoundClass "Sorlag"
	Player.Portrait "PROFSORG"
	Player.ViewHeight 104
	Player.AttackZOffset 51
	Player.MaxHealth 300
	Player.JumpZ 14
	DamageFactor "Slime", 2.0
	DamageFactor "Monsters", 0.5
	DamageFactor "Melee", 0.5
	States
	{
	Spawn:
	  PLAY Z 1
      Loop
    See:
      PLAY ABCDEF 2
	  PLAY G 2 A_SpawnItemEx("Stepskin",0,0,0,0,0,-8)
	  PLAY HIJKLMN 2
	  PLAY O 2 A_SpawnItemEx("Stepskin",0,0,0,0,0,-8)
	  PLAY P 2
      Loop
	See.Walk:
      PLAW ABCDEFGHIJKLMNOPQRSTUVWX 3
      Loop
	RunBack.Walk:
	  PLAW XWVUTSRQPONMLKJIHGFEDCBA 3
      Loop
	RunBack:
	  PLAY QRSTUV 2
	  PLAY W 2 A_SpawnItemEx("Stepskin",0,0,0,0,0,-8)
	  PLAY XY 2
	  PLA2 ABCDE 2
	  PLA2 F 2 A_SpawnItemEx("Stepskin",0,0,0,0,0,-8)
	  PLA2 G 2
      Loop
    JumpLoop:
      PLAJ ABCDEFGHIJ 1 A_CheckFloor("Land")
	  PLAJ K 0 A_JumpIf(waterlevel>=2,"Swim")
	  PLAJ K 1 A_CheckFloor("Land")
      Goto JumpLoop+10
	JumpLoop.Back:
      PLJB ABCDEFGHI 1 A_CheckFloor("Land")
	  PLJB J 0 A_JumpIf(waterlevel>=2,"Swim")
	  PLJB J 1 A_CheckFloor("Land")
      Goto JumpLoop.Back+9
    Land:
      PLAL A 1 A_PlaySound("Player/HeavyLanding",5)
	  PLAL A 1 Radius_Quake(2,10,0,6,0)
	  PLAL BCDEFGHIJ 2 A_JumpIf(momz<-6,"JumpLoop")
	  TNT1 A 0 A_SetUserVar("user_isrunningforward",0)
	  TNT1 A 0 A_SetUserVar("user_iswalkingforward",0)
	  TNT1 A 0 A_SetUserVar("user_isrunningback",0)
	  TNT1 A 0 A_SetUserVar("user_iswalkingback",0)
	  TNT1 A 0 A_SetUserVar("user_isplayerbusy",0)
      Goto Spawn
	Swim:
      PLSW ABCDEFGHIJKL 5 A_JumpIf(waterlevel<=1,"JumpLoop")
      Loop
	Pain:
	  "####" "#" 0 A_SetUserVar("user_isrunningforward",0)
	  "####" "#" 0 A_SetUserVar("user_iswalkingforward",0)
	  "####" "#" 0 A_SetUserVar("user_isrunningback",0)
	  "####" "#" 0 A_SetUserVar("user_iswalkingback",0)
	  "####" "#" 0 A_SetUserVar("user_isplayerbusy",0)
	  "####" "#" 0 A_SetUserVar("user_istaunting",0)
	  "####" "#" 0 SetPlayerProperty(0,0,0)
	  "####" "#" 1 A_Pain
	  Goto Spawn
	AnimTaunt:
	  PLAY Z 0 A_PlaySound("SORTAUNT",2)
	  PLAY Z 1 A_SetUserVar("user_istaunting",0)
	  Goto Spawn
	MeleeAct:
	  SRBT A 0 A_ChangeFlag(NOPAIN,1)
	  SRBT A 0 A_SetUserVar("user_isplayerbusy",1)
	  SRBT A 0 SetPlayerProperty(0,1,0)
	  SRBT ABCEFGH 2
	  SRBT I 2 A_SpawnItemEx("SorlagBite",64,0,32)
	  SRBT JKLMNOP 3
	  SRBT Q 3 SetPlayerProperty(0,0,0)
	  PLAY Z 0 A_SetUserVar("user_isplayerbusy",0)
	  PLAY Z 0 A_ChangeFlag(NOPAIN,0)
	  Goto Spawn
    Death:
	  NADA A 0 A_PlaySound("world/quake",5)
	  NADA A 0 A_PlaySound("world/quake",6)
	  NADA A 0 A_PlaySound("world/quake",7)
	  NADA A 0 A_Jump(128,"Death1")
	  NADD A 3 A_PlayerScream
	  NADD B 3 A_NoBlocking
	  NADD CDEFGHIJKLMNO 3
	  NADD O -1
	  Stop
	Death1:
	  PLD2 A 3 A_PlayerScream
	  PLD2 B 3 A_NoBlocking
	  PLD2 CDEFGHIJKLM 3
	  PLD2 N -1
	  Stop
	}
}
