actor WalkerGun : Weapon
{
  +INVENTORY.UNDROPPABLE
  +WEAPON.CHEATNOTWEAPON
  +WEAPON.NOAUTOAIM
  Weapon.BobStyle Alpha
  Weapon.BobSpeed 1.34
  Weapon.BobRangeX 0.5
  Weapon.BobRangeY 1.4
  Tag "$VWP_WALKERCANNON"
  States
  {
    Spawn:
      NADS A 1 Thing_Remove(0)
      Loop
	Ready:
	  PLGU A 0 A_SetCrosshair(71)
      PLGU A 1 A_WeaponReady
      Loop
	Deselect:
	  PLGU AAAAA 0 A_Lower
      PLGU A 1 A_Lower
      Loop
	Select:
	  PLGU AAAAA 0 A_Raise
      PLGU A 1 A_Raise
      Loop
	Fire:
	  PLGU A 0 A_TakeInventory("VehicleGunReloaded",1)
	  PLGU A 0 A_PlayWeaponSound("Walker/RocketLauncher")
	  PLGU A 0 A_AlertMonsters
	  PLGU A 0 ACS_NamedExecuteAlways("CheckPowerUp",0,0,0,0)
      PLGU A 1 A_FireCustomMissile("WalkerRocket",0,0,-8,24)
	  PLGU A 9
	  PLGU A 0 A_PlayWeaponSound("Walker/RocketLauncher")
	  PLGU A 0 A_AlertMonsters
	  PLGU A 0 ACS_NamedExecuteAlways("CheckPowerUp",0,0,0,0)
      PLGU A 1 A_FireCustomMissile("WalkerRocket",0,0,-8,24)
	  PLGU A 9
	  PLGU A 0 A_PlayWeaponSound("Walker/RocketLauncher")
	  PLGU A 0 A_AlertMonsters
	  PLGU A 0 ACS_NamedExecuteAlways("CheckPowerUp",0,0,0,0)
      PLGU A 1 A_FireCustomMissile("WalkerRocket",0,0,-8,24)
	  PLGU A 9
	  PLGU A 0 A_PlayWeaponSound("Walker/RocketLauncher")
	  PLGU A 0 A_AlertMonsters
	  PLGU A 0 ACS_NamedExecuteAlways("CheckPowerUp",0,0,0,0)
      PLGU A 1 A_FireCustomMissile("WalkerRocket",0,0,-8,24)
	  PLGU A 9
	  PLGU A 0 A_PlayWeaponSound("Walker/RocketLauncher")
	  PLGU A 0 A_AlertMonsters
	  PLGU A 0 ACS_NamedExecuteAlways("CheckPowerUp",0,0,0,0)
      PLGU A 1 A_FireCustomMissile("WalkerRocket",0,0,-8,24)
	  PLGU A 9
	  PLGU A 0 A_PlayWeaponSound("Walker/RocketLauncher")
	  PLGU A 0 A_AlertMonsters
	  PLGU A 0 ACS_NamedExecuteAlways("CheckPowerUp",0,0,0,0)
      PLGU A 1 A_FireCustomMissile("WalkerRocket",0,0,-8,24)
