Actor WalkerNapalmOrb
{
   var int user_burntime;
   
   Height 4
   Radius 2
   Speed 32
   RenderStyle Add
   Alpha 1.0
   Scale 0.25
   Damage (0)
   BounceSound "Walker/NapalmBounce"
   DeathSound "Walker/NapalmExplode"
   DamageType "VehicleDamage"
   PROJECTILE
   -NOGRAVITY
   +BOUNCEONWALLS
   +BRIGHT
   +FORCEXYBILLBOARD
   +DONTSPLASH
   +NODAMAGETHRUST
   +FORCERADIUSDMG
   States
   {
   Spawn:
	  FIRE A 0 A_JumpIf(waterlevel > 1, "Death")
	  FIRE A 1 A_SpawnItemEx("WalkerNapalmTrail",0,0,0)
	  Loop
   Death:
	  NADA A 0 A_PlaySound("Walker/NapalmExplode",6)
      NADA A 0 A_SpawnItemEx("NapalmFireExplosion1",0,0,8,FRandom(2.0,8.0),0,FRandom(1.0,4.0),FRandom(0.0,359.9))
	  NADA A 0 A_SpawnItemEx("NapalmFireExplosion2",0,0,8,FRandom(2.0,8.0),0,FRandom(1.0,4.0),FRandom(0.0,359.9))
	  NADA A 0 A_Explode(150,256,0,0,128)
      Goto Burning
   Burning:
	  NADA A 0 A_JumpIf(user_burntime>=175,"Dissolve")
	  NADA A 0 A_PlaySound("Walker/NapalmBurning",5,1.0,1)
	  NADA A 0 A_Explode(6,256,0,0,128)
	  NADA AA 0 A_SpawnItemEx("NapalmFireFlame",Random(0,192),0,8,FRandom(1.0,4.0),0,FRandom(1.0,2.0),FRandom(0.0,359.9))
	  NADA A 1 A_SetUserVar("user_burntime",user_burntime+1)
	  Loop
   Dissolve:
	  NADA A 1 A_PlaySound("Walker/NapalmDissolve",5)
	  Stop
   }
}
