	  MGUN ABCD 11 A_WeaponReady(WRF_NOBOB|WRF_NOFIRE)
	  MGUN EFGH 10 A_WeaponReady(WRF_NOBOB|WRF_NOFIRE)

	  MGUN A 0 A_GiveInventory("TitanGunSpinning",1)
      Goto AltHold
	Hold:
	  MGUF A 0 A_PlaySound("ToneTitan/HICannonLoop",7,1.0,1)
	  MGUF A 0 A_PlaySound("ToneTitan/HICannonFire",6,1.0,1)
	  MGUF A 0 A_SpawnItemEx("TitanGunFlash",312,80,242,momx,momy,0,0,SXF_SETMASTER)
	  MGUF A 2 A_FireCustomMissile("TitanMGTracer", frandom(-3.0,3.0), 0, 8, 0, 0, frandom(-3.0,3.0))
	  MGUF B 0 A_PlaySound("ToneTitan/HICannonLoop",7,1.0,1)
	  MGUF B 0 A_PlaySound("ToneTitan/HICannonFire",6,1.0,1)
	  MGUF B 2 A_WeaponReady(WRF_NOBOB|WRF_NOFIRE)
	  MGUF C 0 A_PlaySound("ToneTitan/HICannonLoop",7,1.0,1)
	  MGUF C 0 A_PlaySound("ToneTitan/HICannonFire",6,1.0,1)
	  MGUF C 0 A_SpawnItemEx("TitanGunFlash",312,80,242,momx,momy,0,0,SXF_SETMASTER)

	  MGUF D 2 A_WeaponReady(WRF_NOBOB|WRF_NOFIRE)

	  MGUF G 0 A_PlaySound("ToneTitan/HICannonLoop",7,1.0,1)
	  MGUF G 0 A_PlaySound("ToneTitan/HICannonFire",6,1.0,1)

	  MGUF H 2 A_WeaponReady(WRF_NOBOB|WRF_NOFIRE)
	  MGUF H 0 A_Refire
	  MGUF H 0 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  Goto AltHold
	AltHold:
	  MGUN A 0 A_StopSound(6)
	  MGUN A 0 A_PlaySound("ToneTitan/HICannonLoop",7,1.0,1)
	  MGUN A 2 A_WeaponReady(WRF_NOBOB|WRF_NOSECONDARY)
	  MGUN B 0 A_PlaySound("ToneTitan/HICannonLoop",7,1.0,1)
	  MGUN B 2 A_WeaponReady(WRF_NOBOB|WRF_NOSECONDARY)
	  MGUN C 0 A_PlaySound("ToneTitan/HICannonLoop",7,1.0,1)
	  MGUN C 2 A_WeaponReady(WRF_NOBOB|WRF_NOSECONDARY)

	  MGUN G 2 A_WeaponReady(WRF_NOBOB|WRF_NOSECONDARY)
	  MGUN H 0 A_PlaySound("ToneTitan/HICannonLoop",7,1.0,1)
	  MGUN H 2 A_WeaponReady(WRF_NOBOB|WRF_NOSECONDARY)
	  MGUN H 0 A_Refire
	  Goto SpinDown
	SpinDown:
	  MGUN A 0 A_TakeInventory("TitanGunSpinning",1)
	  MGUN A 2 A_PlaySound("ToneTitan/HICannonEnd",7,1.0,0)
	  MGUN BCD 2 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  MGUN EFGH 3 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  MGUN ABCD 4 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  MGUN EFGH 5 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  MGUN ABCD 6 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  MGUN EFGH 7 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)

	  MGUN EFGH 8 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  MGUN ABCD 8 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  MGUN EFGH 9 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  MGUN DCBA 2 A_WeaponReady(WRF_NOBOB|WRF_NOPRIMARY)
	  MGUN A 0 A_GiveInventory("VehicleGunReloaded",1)
      Goto Ready
  }
