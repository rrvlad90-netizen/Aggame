actor TankGun : Weapon
{
  +INVENTORY.UNDROPPABLE
  +WEAPON.CHEATNOTWEAPON
  +WEAPON.DONTBOB
  +WEAPON.NOAUTOAIM
  Tag "$VWP_DISRUPTORCANNON"
  States
  {
    Spawn:
      NADS A 1 Thing_Remove(0)
      Loop
   Ready:
	  PLGU A 0 A_SetCrosshair(71)
      PLGU A 1 A_WeaponReady
      Loop
   Deselect:
	  PLGU AAAAA 0 A_Lower
      PLGU A 1 A_Lower
      Loop
   Select:
	  PLGU AAAAA 0 A_Raise
      PLGU A 1 A_Raise
      Loop
	Fire:
	AltFire:
	  PLGU A 0 A_PlayWeaponSound("Tank/Railfire")
	  PLGU A 0 A_AlertMonsters
	  PLGU A 0 A_SpawnItemEx("GEVFX1",160,0,64)
	  PLGU A 0 ACS_NamedExecuteAlways("CheckPowerUp",0,0,0,0)
	  PLGU A 0 A_TakeInventory("VehicleGunReloaded",1)
	  PLGU A 0 A_Recoil(32)
      PLGU A 1 A_FireCustomMissile("TankBall",0,1,1,0)
	  PLGU B 70
	  PLGU A 0 A_GiveInventory("VehicleGunReloaded",1)
	  PLGU A 0 A_Refire
      Goto Ready
  }
}
