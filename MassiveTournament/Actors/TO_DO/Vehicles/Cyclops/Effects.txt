Actor CyclopsFloorFog
{
	Radius 2
	Height 4
	RenderStyle Add
	Scale 0.0
	Alpha 0.0
	+NOINTERACTION
	+CLIENTSIDEONLY
	+ForceXYBillboard
	+NOTIMEFREEZE
	+NOTONAUTOMAP
	+NONETID
	States
	{
	Spawn:
		NADA A 0 NoDelay A_Jump(170,"Vary2","Vary3")
		EVFG A 0 A_JumpIf(Alpha>=0.65,"SpawnFade")
		EVFG A 0 A_SetScale(ScaleX+0.01)
		EVFG A 1 A_FadeIn(0.05)
		Goto Spawn+1
	SpawnFade:
		EVFG A 0 A_SetScale(ScaleX+0.01)
		EVFG A 1 A_FadeOut(0.025)
		Loop
	Vary2:
		EVFG B 0 A_JumpIf(Alpha>=0.65,"Vary2Fade")
		EVFG B 0 A_SetScale(ScaleX+0.01)
		EVFG B 1 A_FadeIn(0.05)
		Loop
	Vary2Fade:
		EVFG B 0 A_SetScale(ScaleX+0.01)
		EVFG B 1 A_FadeOut(0.025)
		Loop
	Vary3:
		EVFG C 0 A_JumpIf(Alpha>=0.65,"Vary3Fade")
		EVFG C 0 A_SetScale(ScaleX+0.01)
		EVFG C 1 A_FadeIn(0.05)
		Loop
	Vary3Fade:
		EVFG C 0 A_SetScale(ScaleX+0.01)
		EVFG C 1 A_FadeOut(0.025)
		Loop
	}
}
