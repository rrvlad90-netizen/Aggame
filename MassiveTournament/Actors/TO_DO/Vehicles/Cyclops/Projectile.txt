ACTOR CyclopsPlasmaBall : FastProjectile
{
   Speed 160
   Height 4
   Radius 2
   Damage 30
   RenderStyle Add
   Alpha 1.0
   Decal TKScorch
   MissileType "bluefogtrailspawner"
   MissileHeight 4
   DamageType "VehicleDamage"
   PROJECTILE
   +ForceXYBillboard
   +MTHRUSPECIES
   +FORCERADIUSDMG
   +NODAMAGETHRUST
   +BRIGHT
   States
   {
   Spawn:
      FLBL A 1 A_PlaySoundEx("Cycplasmafly",SoundSlot5,1)
      Loop
   Death:
	  PLBL B 0 A_SetScale(3.0)
      PLBL B 0 A_PlaySound("Cyclops/PlasmaHit",5,1.0,0,0.5)
	  PLBL B 0 A_PlaySound("Cyclops/PlasmaHit",6,1.0,0,0.5)
	  PLBL B 1 A_Explode(VH_CYCLOPS_WP_PLASMACANNON_EXPLOSION_DMG,VH_CYCLOPS_WP_PLASMACANNON_EXPLOSION_RADIUS,0,0,VH_CYCLOPS_WP_PLASMACANNON_EXPLOSION_RADIUS/4)
	  Goto FadeOut
   FadeOut:
	  PLBL B 0 A_SetScale(ScaleX-0.1)
	  PLBL B 1 A_FadeOut(0.05)
      Loop
   }
}
