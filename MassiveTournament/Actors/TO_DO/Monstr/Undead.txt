ACTOR UndeadWariorAxe : HMNPCBase
{
	Speed 16
	Health 1500
	Radius 20
	Height 64
	CameraHeight 64
	Mass 500
	PainChance 96
	MeleeRange 80
	MaxDropOffHeight 64
	SeeSound "Undead/Sight"
	ActiveSound "Undead/Idle"
	PainSound "Undead/Pain"
	DeathSound "Undead/Death"
	States
	{
	Spawn:
		NADA A 0 NoDelay ACS_NamedExecuteAlways("SetUndeadActiveTimer",0,0,0,0)
		NADA A 0 A_SpawnItemEx("UndeadSmokeSpawner",0,0,0)
		Goto Idle
	Idle:
		NADA ABCDEFGHIJKLMNOPQRSTUVWXYZ 2 A_LookEx(0,0,0,0,360,"See")
		Loop
	See:
		NADR A 1 A_Chase
		NADR B 0 A_PlaySound("Undead/Step",5,0.6,0,1.3)
		NADR B 0 A_AlertMonsters(1024,AMF_TARGETEMITTER)
		NADR BCDEFGH 1 A_Chase
		NADR I 0 A_PlaySound("Undead/Step",5,0.6,0,1.3)
		NADR I 0 A_AlertMonsters(1024,AMF_TARGETEMITTER)
		NADR IJKLMNO 1 A_Chase
		Loop
	Melee:
	    NADM A 0 A_Jump(64,"ChargeAxe")
	    NADM A 0 A_PlaySound("Undead/Attack")
	    NADM A 0 A_AlertMonsters(1280,AMF_TARGETEMITTER)
		NADM A 2 A_ChangeFlag(NOPAIN,1)
		NADM BCDEFGHIJ 2
		NADM K 2 A_CustomMeleeAttack(random(500,750),"Undead/AxeHit","Undead/AxeHit",DoomslayerBerserk,1)
		NADM LMNOPQRSTUVW 2
		NADM X 2 A_ChangeFlag(NOPAIN,0)
		Goto See
	ChargeAxe:
		N2DM A 0 A_PlaySound("Undead/AxeCharge")
		N2DM A 0 A_ChangeFlag(NOPAIN,1)
		N2DM ABCDE 3
		N2DM F 3 A_PlaySound("Undead/Attack")
		N2DM GHIJ 3
		N2DM K 1 A_FaceTarget
		N2DM K 2 A_SpawnItemEx("UndeadAxeBlast",128,0,0)
		N2DM LMNOPQRSTUVWXY 3
		N2DM Z 3 A_ChangeFlag(NOPAIN,0)
		Goto See
	Pain:
		NADA A 1 A_Pain
		Goto See
	Death:
		NADD A 0 A_ScreamAndUnblock
		NADD ABCDEFGHIJKLMNOPQRSTUVWXYZ 4
		NADD Z 700
		Goto FadeOut
	FadeOut:
		NADD Z 1 A_FadeOut(0.06)
		Loop
	}
}
